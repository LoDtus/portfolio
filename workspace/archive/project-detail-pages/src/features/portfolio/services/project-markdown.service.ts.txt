import { readFile } from "node:fs/promises";
import path from "node:path";
import {
    PROJECTS,
    type ProjectConfig,
} from "@/features/portfolio/constants/profile.constants";

const MARKDOWN_DIR = path.join(process.cwd(), "public", "markdown");

// Đọc tài liệu markdown của project lúc build (server). Tên tệp lấy từ hằng số, không từ đầu vào người dùng.
// Trả về null nếu tệp không tồn tại hoặc rỗng, khi đó project không có trang chi tiết.
export async function getProjectMarkdown(
    markdownFile: string,
): Promise<string | null> {
    try {
        const raw = await readFile(path.join(MARKDOWN_DIR, markdownFile), "utf8");
        return raw.trim().length > 0 ? raw : null;
    } catch (error) {
        console.error("Cannot read project markdown:", markdownFile, error);
        return null;
    }
}

export function findProjectBySlug(slug: string): ProjectConfig | undefined {
    return PROJECTS.find((project) => project.slug === slug);
}

// Các project thật sự có tài liệu để dựng trang chi tiết.
export async function getProjectsWithDetail(): Promise<ProjectConfig[]> {
    const checks = await Promise.all(
        PROJECTS.map(async (project) => {
            if (project.markdownFile === null) return false;
            return (await getProjectMarkdown(project.markdownFile)) !== null;
        }),
    );
    return PROJECTS.filter((_, index) => checks[index]);
}
